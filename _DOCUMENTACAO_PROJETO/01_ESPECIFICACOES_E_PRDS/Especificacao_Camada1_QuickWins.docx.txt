Operação (comprar/alugar) — peso 20 if (profile.operacao &amp;&amp; imovel.operacao === profile.operacao) { score += 20; breakdown.push({ criterio: &apos;Operação&apos;, pontos: 20, max: 20 }); } // 2. Tipo de imóvel — peso 25 if (profile.tipo_imovel) { const pts = imovel.tipo === profile.tipo_imovel ? 25 : tipoProximo(imovel.tipo, profile.tipo_imovel) ? 12 : 0; score += pts; breakdown.push({ criterio: &apos;Tipo de imóvel&apos;, pontos: pts, max: 25 }); } // 3. Localização — peso 20 if (profile.bairros?.length) { const inExact = profile.bairros.includes(imovel.bairro); const pts = inExact ? 20 : 8; score += pts; breakdown.push({ criterio: &apos;Localização&apos;, pontos: pts, max: 20 }); } // 4. Faixa de preço — peso 20 if (profile.preco_max) { const ratio = imovel.preco / profile.preco_max; const pts = ratio &lt;= 1 ? 20 : ratio &lt;= 1.10 ? 15 : ratio &lt;= 1.20 ? 5 : 0; score += pts; breakdown.push({ criterio: &apos;Preço&apos;, pontos: pts, max: 20 }); } // 5. Quartos — peso 10 if (profile.quartos_min) { const diff = imovel.quartos - profile.quartos_min; const pts = diff &gt;= 0 ? 10 : diff === -1 ? 5 : 0; score += pts; breakdown.push({ criterio: &apos;Quartos&apos;, pontos: pts, max: 10 }); } // Normalizar para o total de critérios com dados disponíveis const maxPossible = breakdown.reduce((s, b) =&gt; s + b.max, 0) || 100; const normalized = Math.round((score / maxPossible) * 100); return { score: normalized, percentage: `${normalized}%`, label: normalized &gt;= 85 ? &apos;Ótima combinação&apos; : normalized &gt;= 70 ? &apos;Boa compatibilidade&apos; : normalized &gt;= 50 ? &apos;Compatível&apos; : &apos;Compatibilidade parcial&apos;, color: normalized &gt;= 85 ? &apos;#16A34A&apos; : normalized &gt;= 70 ? &apos;#D97706&apos; : &apos;#DC2626&apos;, breakdown, }; } 3.3 Integração no Template Puppeteer // src/workers/dossie-generator.worker.ts // Modificação no HTML template do dossiê // 1. Mas imobiliárias perdem vendas todo dia porque um lead qualificado — que não encontrou o imóvel certo há 3 semanas — não sabe que o imóvel dos sonhos acaba de entrar no sistema. O alerta proativo inverte essa lógica: o imóvel vai buscar o lead. Este é o recurso com maior impacto financeiro direto para o cliente (imobiliária) porque gera vendas sem nenhuma ação humana. Um lead que esfriou volta quente. 📊 Exemplo Real de Impacto Cenário: 50 leads qualificados no banco de uma imobiliária. • 12 leads buscavam apartamento 3Q em Jardins até R$600k. • Nenhum imóvel compatível havia no momento. • 3 semanas depois: entra apartamento 3Q em Jardins por R$575k. Sem o alerta: O lead só volta se ele mesmo lembrar de checar. Taxa de retorno &lt; 5%. Com o alerta: 12 WhatsApps personalizados em &lt; 30 segundos após o cadastro. Taxa de reabertura de conversa estimada: 40–60%. 4.2 Fluxo Completo — Trigger → Match → Envio TRIGGER: INSERT na tabela &apos;imoveis&apos; (Supabase Webhook ou Postgres trigger) │ ▼ JOB: match-new-property.job.ts 1.
